# The MoOS family of native System Settings modules.
#
# Every directory under modules/ is one KCM plugin:
#
#   modules/<name>/<plugin id>.json   KCM metadata (Name, category, weight, keywords)
#   modules/<name>/ui/main.qml        the page (plus any extra QML it pushes or uses)
#
# The shared C++ backend (src/) is compiled into every plugin, and the shared QML
# (common/) is embedded next to each module's own files, so every page can use
# MoosHero, MoosChip, MoosFactRow ... as plain same-directory types. The qrc prefix
# is /kcm/<plugin id>/ with main.qml at its root: that is where KQuickConfigModule
# looks for a module's page.
#
# A slice that adds a module only adds its directory; nothing here names it.
cmake_minimum_required(VERSION 3.24)
project(moos_settings_kcm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(ECM 6.0 REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Qml Quick)
find_package(KF6 6.0 REQUIRED COMPONENTS CoreAddons KCMUtils)

set(MOOS_KCM_BACKEND_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/src/moosbackend.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/moosbackend.h)
file(GLOB MOOS_KCM_COMMON_QML CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/common/*.qml")

# moos_add_settings_module(<module directory>)
#
# kcmutils_add_qml_kcm() hard-codes one "ui" folder per target; a MoOS module
# needs its own ui/ plus the shared common/ QML in the same qrc directory, so the
# resource file is written here instead.
function(moos_add_settings_module module_dir)
    file(GLOB metadata CONFIGURE_DEPENDS "${module_dir}/*.json")
    list(LENGTH metadata metadata_count)
    if(NOT metadata_count EQUAL 1)
        message(FATAL_ERROR "${module_dir} must hold exactly one <plugin id>.json, found ${metadata_count}")
    endif()
    get_filename_component(plugin_id "${metadata}" NAME_WE)
    if(NOT plugin_id MATCHES "^kcm_moos(_[a-z]+)?$")
        message(FATAL_ERROR "${metadata}: a MoOS module id is kcm_moos or kcm_moos_<word>, not ${plugin_id}")
    endif()
    if(NOT EXISTS "${module_dir}/ui/main.qml")
        message(FATAL_ERROR "${module_dir} has no ui/main.qml")
    endif()

    set(factory "${CMAKE_CURRENT_BINARY_DIR}/${plugin_id}_plugin.cpp")
    set(MOOS_KCM_ID "${plugin_id}")
    set(MOOS_KCM_JSON "${metadata}")
    configure_file("${CMAKE_CURRENT_SOURCE_DIR}/src/plugin.cpp.in" "${factory}" @ONLY)

    kcoreaddons_add_plugin(${plugin_id}
        INSTALL_NAMESPACE "plasma/kcms/systemsettings"
        SOURCES ${factory} ${MOOS_KCM_BACKEND_SOURCES})
    target_include_directories(${plugin_id} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_link_libraries(${plugin_id} PRIVATE
        Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick KF6::CoreAddons KF6::KCMUtils)

    file(GLOB_RECURSE own_files CONFIGURE_DEPENDS "${module_dir}/ui/*")
    set(qrc "<!-- Generated by moos-settings-kcm/CMakeLists.txt; do not edit. -->\n<RCC>\n<qresource prefix=\"/kcm/${plugin_id}\">")
    set(aliases)
    foreach(path ${own_files})
        file(RELATIVE_PATH alias "${module_dir}/ui" "${path}")
        list(APPEND aliases "${alias}")
        string(APPEND qrc "\n<file alias=\"${alias}\">${path}</file>")
    endforeach()
    foreach(path ${MOOS_KCM_COMMON_QML})
        get_filename_component(alias "${path}" NAME)
        if(alias IN_LIST aliases)
            message(FATAL_ERROR "${module_dir}/ui/${alias} shadows the shared common/${alias}")
        endif()
        string(APPEND qrc "\n<file alias=\"${alias}\">${path}</file>")
    endforeach()
    string(APPEND qrc "\n</qresource>\n</RCC>\n")
    set(work "${CMAKE_CURRENT_BINARY_DIR}/_generated_${plugin_id}.qrc")
    set(compiled "${CMAKE_CURRENT_BINARY_DIR}/${plugin_id}.qrc")
    file(WRITE "${work}" "${qrc}")
    execute_process(COMMAND ${CMAKE_COMMAND} -E copy_if_different "${work}" "${compiled}")
    qt_add_resources(resources "${compiled}")
    target_sources(${plugin_id} PRIVATE ${resources})

    set(MOOS_SETTINGS_MODULE_IDS ${MOOS_SETTINGS_MODULE_IDS} ${plugin_id} PARENT_SCOPE)
endfunction()

file(GLOB module_dirs LIST_DIRECTORIES true CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/modules/*")
list(SORT module_dirs)
set(MOOS_SETTINGS_MODULE_IDS)
foreach(module_dir ${module_dirs})
    if(IS_DIRECTORY "${module_dir}")
        moos_add_settings_module("${module_dir}")
    endif()
endforeach()

# The image build loads every module named here, and fails on a missing one. MoOS Themes
# and Mo AI are required too: every theme entry (its launcher shim, the kept pin, Welcome,
# the launcher, Mo AI's themes/appearance/wallpaper) and every Mo AI settings entry (the
# app's gear, moai-config, brain/start, setup-brain) lands on them and has no other page.
foreach(required kcm_moos kcm_moos_update kcm_moos_whatsnew kcm_moos_remote kcm_moos_recovery
                 kcm_moos_appearance kcm_moos_ai)
    if(NOT required IN_LIST MOOS_SETTINGS_MODULE_IDS)
        message(FATAL_ERROR "the MoOS settings family lost ${required}")
    endif()
endforeach()
list(JOIN MOOS_SETTINGS_MODULE_IDS "\n" manifest)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/settings-modules.list" "${manifest}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/settings-modules.list"
        DESTINATION "${KDE_INSTALL_DATADIR}/moos")

option(MOOS_KCM_CONTRACT_CHECK "Build moos-settings-contract-check" OFF)

    add_executable(moos-settings-contract-check src/contractcheck.cpp ${MOOS_KCM_BACKEND_SOURCES})
    target_include_directories(moos-settings-contract-check PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_link_libraries(moos-settings-contract-check PRIVATE
        Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick KF6::CoreAddons KF6::KCMUtils)
endif()
